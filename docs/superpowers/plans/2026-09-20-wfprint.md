# wfprint Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Satu exe `wfprint.exe` dual-mode (`--server`/`--client`) yang ganti SMB sharing dengan raw TCP 9100 di LAN.

**Architecture:** `wfprint.py` dispatch CLI ke `wfprint/server.py` (TcpListener per port, FIFO per printer, tulis via `winspool.drv`) atau `wfprint/client.py` (probe TCP, setup Standard TCP/IP Port via PowerShell). Tanpa driver virtual, tanpa cloud.

**Tech Stack:** Python 3.12 stdlib saja (`socket`, `threading`, `argparse`, `json`, `ctypes`, `subprocess`, `logging`, `unittest`). Freeze `.exe` via PyInstaller ditunda sampai fungsi lolos (ponytail).

**Spec:** `docs/superpowers/specs/2026-09-20-wfprint-design.md`

## Global Constraints

- Windows 10/11 saja, Python 3.12+ di dev, stdlib saja, tanpa dependensi baru.
- Satu printer = satu port TCP, mulai 9100, tambah 1 per printer.
- Bind IP LAN eksplisit, tanpa `0.0.0.0` bila NIC ganda; firewall inbound hanya subnet LAN.
- Timeout terima job 30 dtk; timeout cetak ikut spooler Windows.
- FIFO 1 job aktif per printer; job gagal log + tutup koneksi, server tetap jalan.
- Tanpa auth v1 (LAN dipercaya); tanpa IPP; tanpa driver virtual.

## Review Focus

- Job 0-byte (client buka-tutup tanpa data) harus ditolak tanpa panggil spooler, koneksi ditutup, log `empty job`.
- Job raksasa (~500MB spool PCL) tidak boleh load utuh ke RAM tanpa batas — tulis streaming per chunk 64KB.
- Nama printer unicode/spasi (`"HP LaserJet (Copy 1)"`) harus lolos ctypes wchar tanpa mojibake.
- Port sudah dipakai saat start harus log jelas + exit kode 2, bukan traceback mentah.
- USB dicabut tengah `WritePrinter` harus abort job, tutup koneksi, server terima job berikut.

## File Structure

- Buat: `wfprint/__init__.py` — marker paket kosong.
- Buat: `wfprint/config.py` — load/save `%APPDATA%\wfprint\config.json`, mapping `{"9100": "Nama Printer"}`.
- Buat: `wfprint/spool.py` — `Spool` interface + `FakeSpool` (test) + `WinSpool` (ctypes `winspool.drv`).
- Buat: `wfprint/server.py` — `recv_job`, `handle_conn`, `serve` + lock FIFO per port.
- Buat: `wfprint/client.py` — `probe`, `ps_setup`, `setup`, `print_test_page`.
- Buat: `wfprint.py` — argparse `--server/--client`, dispatch.
- Buat: `tests/test_config.py`, `tests/test_spool.py`, `tests/test_server.py`, `tests/test_client.py` — unittest stdlib.
- Buat: `scripts/install_server.ps1` — firewall inbound LAN + autostart registry + contoh jalan.

```python
import json
import os
from pathlib import Path

DEFAULT_PORT = 9100

def config_dir() -> Path:
    base = os.environ.get("APPDATA") or str(Path.home())
    return Path(base) / "wfprint"

def config_file() -> Path:
    return config_dir() / "config.json"

def load_config() -> dict:
    p = config_file()
    if not p.exists():
        return {}
    return json.loads(p.read_text(encoding="utf-8"))

def save_config(mapping: dict) -> None:
    d = config_dir()
    d.mkdir(parents=True, exist_ok=True)
    config_file().write_text(json.dumps(mapping, indent=2), encoding="utf-8")
```

```python
    ap.add_argument("--driver", default="")

def main(argv=None) -> int:
    ap = build_parser()
    ns = ap.parse_args(argv)
    if ns.server:
        from wfprint import server
        from wfprint import spool
        mapping = {ns.port: ns.printer or "Default"}
        import threading
        stop = threading.Event()
        server.serve(mapping, ns.bind, spool.WinSpool, stop)

    if ns.client:
        from wfprint import client
        ok = client.probe(ns.host, ns.port)
        print("OK" if ok else "FAIL: cek IP/port, firewall server, wfprint --server jalan")
        return 0 if ok else 1
    ap.print_help()
    return 2

if __name__ == "__main__":
    raise SystemExit(main())
```

```python
# wfprint/spool.py
import ctypes
from ctypes import wintypes

class SpoolError(Exception):
    pass

class Spool:
    def open(self, printer_name: str) -> int:
        raise NotImplementedError
    def start_doc(self, handle: int, doc_name: str) -> int:
        raise NotImplementedError
    def write(self, handle: int, data: bytes) -> int:
        raise NotImplementedError
    def end_doc(self, handle: int) -> None:
        raise NotImplementedError
    def close(self, handle: int) -> None:
        raise NotImplementedError
    def abort(self, handle: int) -> None:
        self.close(handle)

class FakeSpool(Spool):
    def __init__(self):
        self.jobs: dict = {}
        self._open: set = set()
    def open(self, printer_name: str) -> int:
        h = abs(hash(printer_name)) % 100000 + 1
        self._open.add(h)
        self.jobs[h] = b""
        return h
    def start_doc(self, handle: int, doc_name: str) -> int:
        if handle not in self._open:
            raise SpoolError("bad handle")
        return 1
    def write(self, handle: int, data: bytes) -> int:
        if handle not in self._open:
            raise SpoolError("bad handle")
        self.jobs[handle] += bytes(data)
        return len(data)
    def end_doc(self, handle: int) -> None:
        if handle not in self._open:
            raise SpoolError("bad handle")
    def close(self, handle: int) -> None:
        self._open.discard(handle)

class _DOCINFO(ctypes.Structure):
    _fields_ = [("pDocName", wintypes.LPWSTR),
                ("pOutputFile", wintypes.LPWSTR),
                ("pDatatype", wintypes.LPWSTR)]

class WinSpool(Spool):
    def __init__(self):
        self._dll = ctypes.WinDLL("winspool.drv")
        self._dll.OpenPrinterW.argtypes = [wintypes.LPWSTR, ctypes.POINTER(wintypes.HANDLE), ctypes.c_void_p]
        self._dll.OpenPrinterW.restype = wintypes.BOOL
        self._dll.StartDocPrinterW.argtypes = [wintypes.HANDLE, wintypes.DWORD, ctypes.POINTER(_DOCINFO)]
        self._dll.StartDocPrinterW.restype = wintypes.DWORD
        self._dll.WritePrinter.argtypes = [wintypes.HANDLE, ctypes.c_void_p, wintypes.DWORD, ctypes.POINTER(wintypes.DWORD)]
        self._dll.WritePrinter.restype = wintypes.BOOL
        self._dll.EndDocPrinter.argtypes = [wintypes.HANDLE]
        self._dll.EndDocPrinter.restype = wintypes.BOOL
        self._dll.AbortPrinter.argtypes = [wintypes.HANDLE]
        self._dll.AbortPrinter.restype = wintypes.BOOL
        self._dll.ClosePrinter.argtypes = [wintypes.HANDLE]
        self._dll.ClosePrinter.restype = wintypes.BOOL

    def open(self, printer_name: str) -> int:
        h = wintypes.HANDLE()
        if not self._dll.OpenPrinterW(printer_name, ctypes.byref(h), None):
            raise SpoolError(f"printer offline: {printer_name}")
        return int(h.value)

    def start_doc(self, handle: int, doc_name: str) -> int:
        info = _DOCINFO(pDocName=doc_name, pOutputFile=None, pDatatype="RAW")
        jid = self._dll.StartDocPrinterW(wintypes.HANDLE(handle), 1, ctypes.byref(info))
        if jid == 0:
            raise SpoolError("StartDocPrinter gagal")
        return int(jid)

    def write(self, handle: int, data: bytes) -> int:
        written = wintypes.DWORD(0)
        buf = ctypes.create_string_buffer(bytes(data))
        if not self._dll.WritePrinter(wintypes.HANDLE(handle), buf, len(data), ctypes.byref(written)):
            raise SpoolError(f"WritePrinter gagal setelah {int(written.value)} byte")
        return int(written.value)

    def end_doc(self, handle: int) -> None:
        self._dll.EndDocPrinter(wintypes.HANDLE(handle))

    def abort(self, handle: int) -> None:
        try:
            self._dll.AbortPrinter(wintypes.HANDLE(handle))
        finally:
            self.close(handle)

    def close(self, handle: int) -> None:
        self._dll.ClosePrinter(wintypes.HANDLE(handle))
```

```python
# wfprint/server.py
import logging
import socket
import threading

RECV_TIMEOUT = 30.0
CHUNK = 65536

log = logging.getLogger("wfprint.server")
_locks: dict = {}
_locks_guard = threading.Lock()

def _lock_for(port: int) -> threading.Lock:
    with _locks_guard:
        return _locks.setdefault(port, threading.Lock())

def recv_job(conn: socket.socket, timeout: float = RECV_TIMEOUT) -> bytes:
    conn.settimeout(timeout)
    parts = []
    try:
        while True:
            b = conn.recv(CHUNK)
            if not b:
                break
            parts.append(b)
    except socket.timeout:
        pass
    return b"".join(parts)

def handle_conn(conn: socket.socket, addr, port: int, printer_name: str, spool) -> int:
    from wfprint.spool import SpoolError
    try:
        data = recv_job(conn)
    except Exception as e:
        log.error("recv gagal port=%s client=%s err=%s", port, addr, e)
        try:
            conn.close()
        except OSError:
            pass
        return 0
    if not data:
        log.warning("empty job port=%s client=%s", port, addr)
        try:
            conn.close()
        except OSError:
            pass
        return 0
    lock = _lock_for(port)
    with lock:
        try:
            h = spool.open(printer_name)
        except SpoolError as e:
            log.error("%s port=%s client=%s", e, port, addr)
            try:
                conn.shutdown(socket.SHUT_RDWR)
            except OSError:
                pass
            try:
                conn.close()
            except OSError:
                pass
            return 0
        try:
            spool.start_doc(h, f"wfprint {addr[0]}:{addr[1]}")
            sent = 0
            for i in range(0, len(data), CHUNK):
                sent += spool.write(h, data[i:i + CHUNK])
            spool.end_doc(h)
            log.info("job ok port=%s printer=%s client=%s bytes=%d", port, printer_name, addr, sent)
            return sent
        except SpoolError as e:
            log.error("cetak gagal port=%s printer=%s client=%s err=%s", port, printer_name, addr, e)
            try:
                spool.abort(h)
            except Exception:
                pass
            return 0
        finally:
            try:
                spool.close(h)
            except Exception:
                pass
            try:
                conn.close()
            except OSError:
                pass

def serve(mapping, bind_ip: str, spool_factory, stop_event) -> None:
    listeners = []
    for port in mapping:
        s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        s.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        try:
            s.bind((bind_ip, int(port)))
        except OSError as e:
            log.error("bind gagal %s:%s err=%s", bind_ip, port, e)
            for x in listeners:
                x.close()
            raise SystemExit(2)
        s.listen(5)
        s.settimeout(0.5)
        listeners.append((int(port), s))
        log.info("listen %s:%s -> %s", bind_ip, port, mapping[port])
    spool = spool_factory()
    try:
        while not stop_event.is_set():
            for port, s in listeners:
                try:
                    conn, addr = s.accept()
                except socket.timeout:
                    continue
                t = threading.Thread(target=handle_conn, args=(conn, addr, port, mapping[port], spool), daemon=True)
                t.start()
    finally:
        for _, s in listeners:
            s.close()
```

```python
# wfprint/client.py
import socket
import subprocess

def probe(host: str, port: int, timeout: float = 3.0) -> bool:
    s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    s.settimeout(timeout)
    try:
        s.connect((host, int(port)))
        return True
    except OSError:
        return False
    finally:
        s.close()

def ps_setup(port_name: str, host: str, port: int, printer_name: str, driver_name: str) -> str:
    pn = printer_name.replace('"', '')
    dn = driver_name.replace('"', '')
    return (
        f'Add-PrinterPort -Name "{port_name}" -PrinterHostAddress "{host}" -PortNumber {int(port)}\n'
        f'Add-Printer -Name "{pn}" -DriverName "{dn}" -PortName "{port_name}"\n'
    )

def setup(host: str, port: int, printer_name: str, driver_name: str) -> int:
    if not probe(host, port):
        print("FAIL: cek IP/port, firewall server, wfprint --server jalan")
        return 1
    port_name = f"TCP-{host}-{int(port)}"
    ps = ps_setup(port_name, host, port, printer_name, driver_name)
    r = subprocess.run(["powershell", "-NoProfile", "-Command", ps], capture_output=True, text=True)
    print(r.stdout)
    if r.returncode != 0:
        print(r.stderr)
    return r.returncode

def print_test_page(printer_name: str) -> int:
    ps = f'rundll32 printui.dll,PrintUIEntry /k /n "{printer_name}"'
    r = subprocess.run(["powershell", "-NoProfile", "-Command", ps], capture_output=True, text=True)
    return r.returncode
```
